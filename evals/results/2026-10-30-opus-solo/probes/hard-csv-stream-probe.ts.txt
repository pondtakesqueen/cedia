function expected(t: string) { try { return { rows: parseCsv(t) } } catch (e: any) { return { err: e } } }
function run(Cls: any, chunks: string[]) {
  const s = new Cls(); const cum: number[] = []; const rows: string[][] = [];
  try { for (const c of chunks) { rows.push(...s.push(c)); cum.push(rows.length) } rows.push(...s.end()); return { rows, cum } }
  catch (e: any) { return { err: e, rows, cum } }
}
function check(t: string, chunks: string[]) {
  cases++;
  const exp = expected(t); const got: any = run(CsvStream, chunks); const ref: any = run(Ref, chunks);
  if (exp.err) {
    if (!(got.err instanceof SyntaxError)) { fails.push(JSON.stringify({ t, chunks, got: got.rows, err: String(got.err) })); return }
    if (got.err.message !== exp.err.message) msgDiff++;
  } else {
    if (got.err || JSON.stringify(got.rows) !== JSON.stringify(exp.rows)) { fails.push(JSON.stringify({ t, chunks, got: got.rows, err: String(got.err) })); return }
    if (JSON.stringify(got.cum) !== JSON.stringify(ref.cum)) { eagerFail++; if (eagerFail < 3) console.log("eager", JSON.stringify({ chunks, got: got.cum, ref: ref.cum })) }
  }
}
